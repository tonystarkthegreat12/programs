import pandas as pd
import numpy as np

from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import accuracy_score

# Step 1: Load dataset from CSV file
df = pd.read_csv("Naive-Bayes-Classification-Data.csv")

# Display dataset
print("Dataset:\n")
print(df)

# Step 2: Split dataset into features (X) and target (Y)
X = df.iloc[:, :-1]   # All columns except last
Y = df.iloc[:, -1]    # Last column

# Step 3: Split data into Training (80%) and Testing (20%)
X_train, X_test, Y_train, Y_test = train_test_split(
    X,
    Y,
    test_size=0.2,
    random_state=42
)

# Step 4: Create Naive Bayes model
model = GaussianNB()

# Step 5: Train the model
model.fit(X_train, Y_train)

# Step 6: Predict test data
Y_pred = model.predict(X_test)

# Step 7: Compute Accuracy
accuracy = accuracy_score(Y_test, Y_pred)

print("\nPredicted Values:\n")
print(Y_pred)

print("\nActual Values:\n")
print(Y_test.values)

print(f"\nNaive Bayes Classifier Accuracy: {accuracy:.2f}")

# Step 8: Test with a New Sample
# Example:
# Glucose = 130
# BloodPressure = 85

new_sample = pd.DataFrame(
    [[130, 85]],
    columns=X.columns
)

prediction = model.predict(new_sample)

# Step 9: Display Prediction
if prediction[0] == 1:
    print("\nPrediction for New Sample: Diabetes")
else:
    print("\nPrediction for New Sample: No Diabetes")
